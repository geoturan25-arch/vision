Yenilenmiş Katalog Çizim Fonksiyonu function renderCatalog(){ const t = T[lang]; let list = getAllProducts(); // Kategori Filtresi if(catalogType !== 'all'){ list = list.filter(p => p.type === catalogType); } // Marka Filtresi if(catalogBrand !== 'all'){ list = list.filter(p => p.brand === catalogBrand); } // Arama Kelimesi Filtresi if(catalogSearchQuery){ list = list.filter(p => p.name.toLowerCase().includes(catalogSearchQuery) || (p.res && p.res.toLowerCase().includes(catalogSearchQuery)) || (p.spec && p.spec.toLowerCase().includes(catalogSearchQuery)) ); } if(list.length === 0){ $("cards").innerHTML = `
Aradığınız kriterlere uygun ürün bulunamadı.
`; return; } $("cards").innerHTML = list.map(p => { const isSelected = selectedProductIds.has(p.id); const imgHtml = p.image ? `` : (ICONS[p.icon] || ICONS['bullet']); return `
${t.badge_sel}
${getBrandBadge(p.brand)}
${imgHtml}

${p.name}

Özellik${p.res || p.spec || '-'}
Donanım${p.lens || '-'}
Gece/Depolama${p.ir || '-'}
